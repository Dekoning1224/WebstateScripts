function Write-Log {

    param(
        [Parameter(Mandatory)]
        [string]$Message
    )

    $Line = (
        "$(Get-Date -Format 'yyyy-MM-dd HH:mm:ss') " +
        "- MANUALREBOOT - $Message"
    )

    Add-Content `
        -Path $LogFile `
        -Value $Line `
        -Encoding UTF8

    Write-Host $Line
}


function Get-TextFileValue {

    param(
        [Parameter(Mandatory)]
        [string]$Path
    )

    if (-not (Test-Path $Path)) {
        return ''
    }

    try {

        return (
            Get-Content `
                -Path $Path `
                -Raw `
                -ErrorAction Stop
        ).Trim()

    }
    catch {

        return ''
    }
}


function Get-WtsStateName {

    param(
        [int]$State
    )

    switch ($State) {

        0 { 'Active' }
        1 { 'Connected' }
        2 { 'ConnectQuery' }
        3 { 'Shadow' }
        4 { 'Disconnected' }
        5 { 'Idle' }
        6 { 'Listen' }
        7 { 'Reset' }
        8 { 'Down' }
        9 { 'Init' }

        default {
            "Unknown($State)"
        }
    }
}


function ConvertTo-LdapFilterValue {

    param(
        [Parameter(Mandatory)]
        [string]$Value
    )

    $Builder = New-Object System.Text.StringBuilder

    foreach ($Character in $Value.ToCharArray()) {

        switch ([int][char]$Character) {

            0 {
                [void]$Builder.Append('\00')
            }

            40 {
                [void]$Builder.Append('\28')
            }

            41 {
                [void]$Builder.Append('\29')
            }

            42 {
                [void]$Builder.Append('\2a')
            }

            92 {
                [void]$Builder.Append('\5c')
            }

            default {
                [void]$Builder.Append($Character)
            }
        }
    }

    return $Builder.ToString()
}

# We eisen:
#
#   CBS package = UninstallPending

#   CBS package bestaat niet meer EN Get-HotFix ziet KB ook niet meer.

#

                'CBS bevestigt UninstallPending. ' +
                'Rollback is klaar voor reboot.'

public class WebstateWtsSessionV4

    [StructLayout(LayoutKind.Sequential)]

        "wtsapi32.dll",

        CharSet = CharSet.Unicode,

        int sessionId,

    );

    private static extern void WTSFreeMemory(
        IntPtr pMemory
    );

        int sessionId,
        WTS_INFO_CLASS infoClass
    )

        IntPtr buffer = IntPtr.Zero;
        int bytes = 0;

        try
        {
            if (!WTSQuerySessionInformation(
                IntPtr.Zero,
                sessionId,
                infoClass,
                out buffer,
                out bytes))
            {
                return "";
            }

            if (buffer == IntPtr.Zero)
                return "";

            return Marshal.PtrToStringUni(buffer) ?? "";
        }
        finally
        {
            if (buffer != IntPtr.Zero)
                WTSFreeMemory(buffer);
        }
    }

        int count = 0;

        List<WebstateWtsSessionV4> result =

                }

                current += size;
            }
        }
        finally
        {
            if (buffer != IntPtr.Zero)
                WTSFreeMemory(buffer);
        }

        return result.ToArray();
    }

function Get-WebstateDomainContext {

    $RootDse     = $null
    $DomainEntry = $null
    $Partitions  = $null
    $Searcher    = $null

    try {

        $RootDse = New-Object `
            System.DirectoryServices.DirectoryEntry(
                'LDAP://RootDSE'
            )


        $DefaultNamingContext = [string](
            $RootDse.Properties[
                'defaultNamingContext'
            ][0]
        )


        $ConfigurationNamingContext = [string](
            $RootDse.Properties[
                'configurationNamingContext'
            ][0]
        )


        if (-not $DefaultNamingContext) {

            throw 'defaultNamingContext ontbreekt.'
        }


        $DomainEntry = New-Object `
            System.DirectoryServices.DirectoryEntry(
                "LDAP://$DefaultNamingContext"
            )


        $DomainSidBytes =
            $DomainEntry.Properties[
                'objectSid'
            ][0]


        if (-not $DomainSidBytes) {

            throw 'Domain SID ontbreekt.'
        }


        $DomainSid = New-Object `
            System.Security.Principal.SecurityIdentifier(
                $DomainSidBytes,
                0
            )


        $NetBIOSDomain = ''


        if ($ConfigurationNamingContext) {

            try {

                $Partitions = New-Object `
                    System.DirectoryServices.DirectoryEntry(
                        "LDAP://CN=Partitions,$ConfigurationNamingContext"
                    )


                $Searcher = New-Object `
                    System.DirectoryServices.DirectorySearcher(
                        $Partitions
                    )


                $EscapedNC =
                    ConvertTo-LdapFilterValue `
                        -Value $DefaultNamingContext


                $Searcher.Filter =
                    "(&(objectClass=crossRef)(nCName=$EscapedNC))"


                [void]$Searcher.PropertiesToLoad.Add(
                    'nETBIOSName'
                )


                $Result =
                    $Searcher.FindOne()


                if (
                    $Result -and
                    $Result.Properties[
                        'netbiosname'
                    ].Count -gt 0
                ) {

                    $NetBIOSDomain =
                        [string]$Result.Properties[
                            'netbiosname'
                        ][0]
                }

            }
            catch {

                $NetBIOSDomain = ''
            }
        }


        try {

            $DnsDomain = (
                [System.DirectoryServices.ActiveDirectory.Domain]::GetComputerDomain()
            ).Name

        }
        catch {

            $DnsDomain = ''
        }


        return [pscustomobject]@{

            DefaultNamingContext =
                $DefaultNamingContext

            DnsDomain =
                $DnsDomain

            NetBIOSDomain =
                $NetBIOSDomain

            DomainSid =
                $DomainSid.Value

            DomainAdminsSid =
                "$($DomainSid.Value)-512"
        }

    }
    finally {

        if ($Searcher) {
            $Searcher.Dispose()
        }

        if ($Partitions) {
            $Partitions.Dispose()
        }

        if ($DomainEntry) {
            $DomainEntry.Dispose()
        }

        if ($RootDse) {
            $RootDse.Dispose()
        }
    }
}


# =====================================================================
# DOMAIN ADMIN DETECTIE
#
# Geen tokenGroups.
#
# User:
#   primaryGroupID=512
# OF
#   memberOf recursief → group SID = DOMAIN-SID-512
# =====================================================================

function Test-IsDomainAdmin {

    param(

        [Parameter(Mandatory)]
        [string]$Domain,

        [Parameter(Mandatory)]
        [string]$UserName,

        [Parameter(Mandatory)]
        $DomainContext
    )


    $SearchRoot   = $null
    $UserSearcher = $null


    try {

        $DomainMatches = $false


        if (
            $DomainContext.NetBIOSDomain -and
            $Domain -ieq $DomainContext.NetBIOSDomain
        ) {

            $DomainMatches = $true
        }


        if (
            $DomainContext.DnsDomain -and
            $Domain -ieq $DomainContext.DnsDomain
        ) {

            $DomainMatches = $true
        }


        if (-not $DomainMatches) {

            return $false
        }


        $SearchRoot = New-Object `
            System.DirectoryServices.DirectoryEntry(
                "LDAP://$($DomainContext.DefaultNamingContext)"
            )


        $EscapedUser =
            ConvertTo-LdapFilterValue `
                -Value $UserName


        $UserSearcher = New-Object `
            System.DirectoryServices.DirectorySearcher(
                $SearchRoot
            )


        $UserSearcher.Filter = (
            '(&' +
                '(objectCategory=person)' +
                '(objectClass=user)' +
                "(sAMAccountName=$EscapedUser)" +
            ')'
        )


        [void]$UserSearcher.PropertiesToLoad.Add(
            'primaryGroupID'
        )

        [void]$UserSearcher.PropertiesToLoad.Add(
            'memberOf'
        )


        $UserResult =
            $UserSearcher.FindOne()


        if (-not $UserResult) {

            return $false
        }


        if (
            $UserResult.Properties[
                'primarygroupid'
            ].Count -gt 0
        ) {

            if (
                [int]$UserResult.Properties[
                    'primarygroupid'
                ][0] -eq 512
            ) {

                return $true
            }
        }


        $Queue =
            New-Object `
                'System.Collections.Generic.Queue[string]'


        $Visited =
            New-Object `
                'System.Collections.Generic.HashSet[string]' `
                ([System.StringComparer]::OrdinalIgnoreCase)


        foreach (
            $GroupDN in
            $UserResult.Properties[
                'memberof'
            ]
        ) {

            if ($GroupDN) {

                $Queue.Enqueue(
                    [string]$GroupDN
                )
            }
        }


        $ProcessedGroups = 0


        while (
            $Queue.Count -gt 0
        ) {

            $GroupDN =
                $Queue.Dequeue()


            if (
                -not $Visited.Add(
                    $GroupDN
                )
            ) {

                continue
            }


            $ProcessedGroups++


            if (
                $ProcessedGroups -gt 1000
            ) {

                throw 'Meer dan 1000 groepen gevolgd.'
            }


            $GroupSearcher = $null


            try {

                $GroupSearcher = New-Object `
                    System.DirectoryServices.DirectorySearcher(
                        $SearchRoot
                    )


                $EscapedGroupDN =
                    ConvertTo-LdapFilterValue `
                        -Value $GroupDN


                $GroupSearcher.Filter = (
                    '(&' +
                        '(objectCategory=group)' +
                        "(distinguishedName=$EscapedGroupDN)" +
                    ')'
                )


                [void]$GroupSearcher.PropertiesToLoad.Add(
                    'objectSid'
                )

                [void]$GroupSearcher.PropertiesToLoad.Add(
                    'memberOf'
                )


                $GroupResult =
                    $GroupSearcher.FindOne()


                if (-not $GroupResult) {

                    continue
                }


                if (
                    $GroupResult.Properties[
                        'objectsid'
                    ].Count -gt 0
                ) {

                    $GroupSid = New-Object `
                        System.Security.Principal.SecurityIdentifier(
                            $GroupResult.Properties[
                                'objectsid'
                            ][0],
                            0
                        )


                    if (
                        $GroupSid.Value -eq
                        $DomainContext.DomainAdminsSid
                    ) {

                        return $true
                    }
                }


                foreach (
                    $ParentDN in
                    $GroupResult.Properties[
                        'memberof'
                    ]
                ) {

                    if ($ParentDN) {

                        $Queue.Enqueue(
                            [string]$ParentDN
                        )
                    }
                }

            }
            finally {

                if ($GroupSearcher) {
                    $GroupSearcher.Dispose()
                }
            }
        }


        return $false

    }
    catch {

        Write-Log (
            "WAARSCHUWING: Domain Admin-status van " +
            "$Domain\$UserName kon niet betrouwbaar worden vastgesteld: " +
            "$($_.Exception.Message)"
        )


        return $false

    }
    finally {

        if ($UserSearcher) {
            $UserSearcher.Dispose()
        }

        if ($SearchRoot) {
            $SearchRoot.Dispose()
        }
    }
}

    $SafeToReboot = $true

}
else {

    $NonDisconnected = @(
        $RelevantSessions |
        Where-Object {
            $_.State -ne 4
        }
    )


    if ($NonDisconnected.Count -gt 0) {

        foreach ($Session in $NonDisconnected) {

            Write-Log (
                "GEEN REBOOT: $($Session.FullUser), " +
                "Session=$($Session.SessionId), " +
                "State=$($Session.StateName) is niet Disconnected."
            )
        }


        $SafeToReboot = $false

    }
    else {

        try {

            $DisconnectEvents = @(
                Get-WinEvent `
                    -FilterHashtable @{
                        LogName   = 'Microsoft-Windows-TerminalServices-LocalSessionManager/Operational'
                        Id        = 24
                        StartTime = (Get-Date).AddDays(-30)
                    } `
                    -ErrorAction Stop
            )

        }
        catch {

            Write-Log (
                'GEEN REBOOT: disconnect-eventlog kon niet worden gelezen.'
            )

            exit 1
        }


        $Now          = Get-Date
        $AllOldEnough = $true


        foreach ($Session in $RelevantSessions) {

            $DisconnectTime = $null


            foreach ($Event in $DisconnectEvents) {

                try {

                    [xml]$Xml =
                        $Event.ToXml()


                    # Event ID 24 van TerminalServices-LocalSessionManager
                    # bewaart User en SessionID op deze servers onder
                    # UserData/EventXML, niet onder EventData. Gebruik XPath
                    # met local-name() zodat de parsing niet afhankelijk is
                    # van XML namespaces of de Windows-weergavetaal.

                    $SessionIdNode =
                        $Xml.SelectSingleNode(
                            "/*[local-name()='Event']/*[local-name()='UserData']/*[local-name()='EventXML']/*[local-name()='SessionID']"
                        )


                    if (-not $SessionIdNode) {
                        continue
                    }


                    if (
                        [int]$SessionIdNode.InnerText -ne
                        $Session.SessionId
                    ) {

                        continue
                    }


                    $EventUserNode =
                        $Xml.SelectSingleNode(
                            "/*[local-name()='Event']/*[local-name()='UserData']/*[local-name()='EventXML']/*[local-name()='User']"
                        )


                    if (-not $EventUserNode) {
                        continue
                    }


                    $EventUser =
                        [string]$EventUserNode.InnerText


                    if (
                        $EventUser -and
                        $EventUser -ine $Session.FullUser -and
                        $EventUser -ine $Session.UserName
                    ) {

                        continue
                    }


                    $DisconnectTime =
                        $Event.TimeCreated


                    break

                }
                catch {

                    continue
                }
            }


            if (-not $DisconnectTime) {

                Write-Log (
                    "GEEN REBOOT: disconnect-tijd van " +
                    "$($Session.FullUser) kon niet betrouwbaar worden vastgesteld."
                )


                $AllOldEnough = $false

                continue
            }


            $DisconnectedMinutes = [math]::Floor(
                ($Now - $DisconnectTime).TotalMinutes
            )


            Write-Log (
                "DISCONNECTED: $($Session.FullUser), " +
                "Session=$($Session.SessionId), " +
                "duur=$DisconnectedMinutes minuten."
            )


            if (
                $DisconnectedMinutes -lt
                $MinimumDisconnectedMinutes
            ) {

                Write-Log (
                    "GEEN REBOOT: $($Session.FullUser) is pas " +
                    "$DisconnectedMinutes minuten disconnected."
                )


                $AllOldEnough = $false
            }
        }


        if ($AllOldEnough) {

            Write-Log (
                'Alle niet-Domain-Admin gebruikers zijn minimaal ' +
                "$MinimumDisconnectedMinutes minuten Disconnected."
            )


            Write-Log 'DIRECTE REBOOT TOEGESTAAN.'

            $SafeToReboot = $true
        }
    }
}


if (-not $SafeToReboot) {

    Write-Host ''
    Write-Host '============================================================' -ForegroundColor Red
    Write-Host ' GEEN REBOOT - ER ZIJN BLOKKERENDE USERSESSIES' -ForegroundColor Red
    Write-Host '============================================================' -ForegroundColor Red
    Write-Host ''

    exit 2
}


if ($CheckOnly) {

    Write-Host ''
    Write-Host '============================================================' -ForegroundColor Green
    Write-Host ' REBOOT IS OP DIT MOMENT TOEGESTAAN' -ForegroundColor Green
    Write-Host '============================================================' -ForegroundColor Green
    Write-Host ''
    Write-Host 'CBS rollback is gecontroleerd.'
    Write-Host 'Usersessies zijn gecontroleerd.'
    Write-Host 'Er is niets gewijzigd.'
    Write-Host ''

    exit 0
}

$PostBootContent = @'
$ErrorActionPreference = 'SilentlyContinue'
$ProgressPreference    = 'SilentlyContinue'

$BaseDir = 'C:\ProgramData\Webstate\WindowsUpdateGuard'

$LogFile        = Join-Path $BaseDir 'rollback.log'
$RollbackFile   = Join-Path $BaseDir 'rollback-state.txt'
$RollbackKBFile = Join-Path $BaseDir 'rollback-kb.txt'
$RebootFile     = Join-Path $BaseDir 'rollback-reboot.txt'

$InstalledFile  = Join-Path $BaseDir 'installed.txt'
$ExitCodeFile   = Join-Path $BaseDir 'exitcode.txt'
$StateFile      = Join-Path $BaseDir 'state.txt'
$KBFile         = Join-Path $BaseDir 'targetkb.txt'

$GuardUrl  = 'http://provisioning.webstate.nl/snom/_D7/WindowsUpdateKBGuard.ps1.txt'
$GuardFile = 'C:\ProgramData\Webstate\WindowsUpdateGuard.ps1'

$TaskName       = 'Webstate - KB Guard after rollback'
$RebootTaskName = 'Webstate - Reboot after Windows Update rollback'


function Write-PostLog {

    param([string]$Message)

    Add-Content `
        -Path $LogFile `
        -Value "$(Get-Date -Format 'yyyy-MM-dd HH:mm:ss') - POSTBOOT - $Message" `
        -Encoding UTF8
}


Write-PostLog 'Post-boot rollback-verificatie gestart.'

Start-Sleep -Seconds 120


$ExpectedKB = ''

if (Test-Path $RollbackKBFile) {

    $ExpectedKB = (
        Get-Content `
            $RollbackKBFile `
            -Raw `
            -ErrorAction SilentlyContinue
    ).Trim()
}


$Success = $false


for ($Attempt = 1; $Attempt -le 10; $Attempt++) {

    Write-PostLog "Guard/verificatie poging $Attempt van 10."


    try {

        Invoke-RestMethod `
            -Uri $GuardUrl `
            -OutFile $GuardFile `
            -ErrorAction Stop


        Write-PostLog 'Laatste WindowsUpdateKBGuard gedownload.'


        Remove-Item `
            $ExitCodeFile,
            $InstalledFile,
            $StateFile,
            $KBFile `
            -Force `
            -ErrorAction SilentlyContinue


        & $GuardFile -Mode Guard


        Start-Sleep -Seconds 5


        if (-not (Test-Path $ExitCodeFile)) {
            throw 'exitcode.txt ontbreekt.'
        }

        if (-not (Test-Path $InstalledFile)) {
            throw 'installed.txt ontbreekt.'
        }

        if (-not (Test-Path $StateFile)) {
            throw 'state.txt ontbreekt.'
        }

        if (-not (Test-Path $KBFile)) {
            throw 'targetkb.txt ontbreekt.'
        }


        $ExitCode = (
            Get-Content $ExitCodeFile -Raw
        ).Trim()


        $Installed = (
            Get-Content $InstalledFile -Raw
        ).Trim()


        $GuardState = (
            Get-Content $StateFile -Raw
        ).Trim()


        $GuardKB = (
            Get-Content $KBFile -Raw
        ).Trim()


        Write-PostLog (
            "Verification: " +
            "ExpectedKB=$ExpectedKB " +
            "GuardKB=$GuardKB " +
            "ExitCode=$ExitCode " +
            "Installed=$Installed " +
            "GuardState=$GuardState"
        )


        if (
            $ExpectedKB -and
            $GuardKB -and
            $ExpectedKB -ine $GuardKB
        ) {

            throw (
                "KB mismatch: Expected=$ExpectedKB Guard=$GuardKB"
            )
        }


        if (
            $ExitCode -eq '0' -and
            $Installed -eq '0'
        ) {

            Set-Content `
                -Path $RollbackFile `
                -Value 'OK' `
                -Encoding ASCII


            Set-Content `
                -Path $RebootFile `
                -Value "COMPLETED: $(Get-Date -Format 'yyyy-MM-dd HH:mm:ss')" `
                -Encoding ASCII


            Write-PostLog "$GuardKB is niet meer geïnstalleerd."
            Write-PostLog 'rollback-state.txt = OK.'
            Write-PostLog "Guard state = $GuardState."


            $Success = $true

            break
        }


        Write-PostLog 'Verificatie nog niet OK.'

    }
    catch {

        Write-PostLog (
            "Poging $Attempt fout: $($_.Exception.Message)"
        )
    }


    if ($Attempt -lt 10) {

        Start-Sleep -Seconds 60
    }
}


if ($Success) {

    Disable-ScheduledTask `
        -TaskName $TaskName `
        -ErrorAction SilentlyContinue |
    Out-Null


    Unregister-ScheduledTask `
        -TaskName $RebootTaskName `
        -Confirm:$false `
        -ErrorAction SilentlyContinue


    Write-PostLog 'Rollback volledig afgerond.'
}
else {

    Set-Content `
        -Path $RollbackFile `
        -Value 'PostBootVerificationFailed' `
        -Encoding ASCII


    Write-PostLog (
        'ERROR: probleem-KB kon niet als verwijderd worden bevestigd.'
    )
}
'@


Set-Content `
    -Path $PostBootScript `
    -Value $PostBootContent `
    -Encoding UTF8


Write-Log 'Post-boot finalizer geschreven.'

$PostBootArguments = (

)


$PostBootAction =
    New-ScheduledTaskAction `

if (

    Get-TextFileValue `

Set-Content `

    -Value 'RebootingNow' `

    }

    )
